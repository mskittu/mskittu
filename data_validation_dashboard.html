# validation_engine/test_generator.py
# AI-powered test case generation using Claude API

import anthropic
import json
from dataclasses import dataclass
from typing import List, Dict
from .schema_reader import SchemaReader

@dataclass
class TestCase:
  id: str
  name: str
  check_type: str
  expression: str
  expected: str

class TestCaseGenerator:
  """Uses Claude API to dynamically generate validation test cases
  based on schema comparison between source and target."""

  def __init__(self, source_config: dict, target_config: dict):
    self.client = anthropic.Anthropic()
    self.source = SchemaReader(source_config)
    self.target = SchemaReader(target_config)

  def generate(self) -> List[TestCase]:
    src_schema = self.source.read()
    tgt_schema = self.target.read()
    schema_diff = self._compute_diff(src_schema, tgt_schema)

    prompt = f"""You are a data quality expert.
    Given this schema diff between source and target:
    {json.dumps(schema_diff, indent=2)}
    Generate comprehensive validation test cases.
    Return ONLY a JSON array of test objects with:
    id, name, check_type, expression, expected"""

    resp = self.client.messages.create(
      model="claude-sonnet-4-20250514",
      max_tokens=2000,
      messages=[{"role": "user", "content": prompt}]
    )
    raw = json.loads(resp.content[0].text)
    return [TestCase(**tc) for tc in raw]

  def _compute_diff(self, src, tgt) -> dict:
    added = [c for c in src if c not in tgt]
    removed = [c for c in tgt if c not in src]
    type_diffs = {k: (src[k], tgt[k]) for k in src
                  if k in tgt and src[k] != tgt[k]}
    return {"added": added, "removed": removed, "type_diffs": type_diffs}

# Oracle database connector using cx_Oracle

import cx_Oracle
import pandas as pd
from typing import Optional
import os

class OracleConnector:
  """Connects to Oracle source systems and extracts
  schema metadata and row-level data."""

  def __init__(self):
    self.dsn = cx_Oracle.makedsn(
      os.getenv("ORACLE_HOST"),
      os.getenv("ORACLE_PORT", 1521),
      service_name=os.getenv("ORACLE_SERVICE")
    )
    self.conn = None

  def connect(self):
    self.conn = cx_Oracle.connect(
      user=os.getenv("ORACLE_USER"),
      password=os.getenv("ORACLE_PASS"),
      dsn=self.dsn
    )

  def get_schema(self, table: str) -> pd.DataFrame:
    """Returns column name, data type, nullable for a table."""
    sql = """
      SELECT COLUMN_NAME, DATA_TYPE, NULLABLE
      FROM ALL_TAB_COLUMNS
      WHERE TABLE_NAME = :tbl
      ORDER BY COLUMN_ID"""
    return pd.read_sql(sql, self.conn, params={"tbl": table.upper()})

  def get_row_count(self, table: str) -> int:
    cur = self.conn.cursor()
    cur.execute(f"SELECT COUNT(*) FROM {table}")
    return cur.fetchone()[0]

  def get_null_counts(self, table: str, col: str) -> int:
    cur = self.conn.cursor()
    cur.execute(f"SELECT COUNT(*) FROM {table} WHERE {col} IS NULL")
    return cur.fetchone()[0]

  def close(self):
    if self.conn: self.conn.close()

# Apache Hive connector using PyHive

from pyhive import hive
import pandas as pd
from pyspark.sql import SparkSession
import os

class HiveConnector:
  """Connects to Hive Metastore and executes HQL queries.
  Also supports SparkSession for large-scale operations."""

  def __init__(self, use_spark: bool = False):
    self.host = os.getenv("HIVE_HOST", "hiveserver2")
    self.port = int(os.getenv("HIVE_PORT", 10000))
    self.database = os.getenv("HIVE_DB", "default")
    self.use_spark = use_spark
    self._spark = None

  @property
  def spark(self) -> SparkSession:
    if not self._spark:
      self._spark = SparkSession.builder \
        .appName("DataValidation") \
        .enableHiveSupport() \
        .getOrCreate()
    return self._spark

  def get_schema(self, table: str) -> pd.DataFrame:
    conn = hive.Connection(host=self.host, port=self.port,
                           database=self.database)
    return pd.read_sql(f"DESCRIBE {table}", conn)

  def read_table_spark(self, table: str):
    """Read full table as Spark DataFrame for large jobs."""
    return self.spark.sql(f"SELECT * FROM {self.database}.{table}")

  def get_duplicate_count(self, table: str, pk_col: str) -> int:
    df = self.read_table_spark(table)
    from pyspark.sql.functions import count
    dupes = df.groupBy(pk_col).count().filter("count > 1")
    return dupes.count()

# Azure Data Lake Storage Gen2 connector

from azure.storage.filedatalake import DataLakeServiceClient
from azure.identity import DefaultAzureCredential
from pyspark.sql import SparkSession
import os

class AzureADLSConnector:
  """Azure Data Lake Gen2 connector using Managed Identity.
  Supports Spark-native reads for large validation jobs."""

  def __init__(self):
    self.account = os.getenv("ADLS_ACCOUNT")
    self.container = os.getenv("ADLS_CONTAINER")
    credential = DefaultAzureCredential()
    self.client = DataLakeServiceClient(
      account_url=f"https://{self.account}.dfs.core.windows.net",
      credential=credential
    )

  def read_parquet_spark(self, path: str):
    """Read Parquet data from ADLS into Spark DataFrame."""
    spark = SparkSession.builder.getOrCreate()
    full_path = (
      f"abfss://{self.container}@"
      f"{self.account}.dfs.core.windows.net/{path}"
    )
    return spark.read.parquet(full_path)

  def write_report(self, report_json: str, run_id: str):
    """Write validation report JSON back to ADLS."""
    fs = self.client.get_file_system_client(self.container)
    file_client = fs.get_file_client(f"reports/{run_id}.json")
    file_client.upload_data(report_json, overwrite=True)
# validation_engine/row_count.py
# PySpark-based validation checks

from pyspark.sql import DataFrame
from pyspark.sql import functions as F
from dataclasses import dataclass

@dataclass
class CheckResult:
  check: str
  status: str # PASS | FAIL | WARN
  detail: str

def validate_row_count(src: DataFrame, tgt: DataFrame) -> CheckResult:
  sc, tc = src.count(), tgt.count()
  pct_diff = abs(sc - tc) / max(sc, 1)
  if sc == tc:
    return CheckResult("row_count", "PASS", f"Both={sc:,}")
  elif pct_diff < 0.001:
    return CheckResult("row_count", "WARN", f"src={sc:,} tgt={tc:,}")
  return CheckResult("row_count", "FAIL", f"src={sc:,} tgt={tc:,}")

def validate_nulls(df: DataFrame, col: str,
                    threshold: float = 0.05) -> CheckResult:
  total = df.count()
  nulls = df.filter(F.col(col).isNull()).count()
  pct = nulls / max(total, 1)
  status = "PASS" if pct == 0 else ("WARN" if pct < threshold else "FAIL")
  return CheckResult(f"null_{col}", status, f"{nulls:,} nulls ({pct:.2%})")

def validate_pk_integrity(df: DataFrame, pk: str) -> CheckResult:
  total = df.count()
  distinct = df.select(pk).distinct().count()
  dupes = total - distinct
  status = "PASS" if dupes == 0 else "FAIL"
  return CheckResult(f"pk_{pk}", status, f"{dupes} duplicates")

def validate_dtype_match(src: DataFrame, tgt: DataFrame) -> list[CheckResult]:
  src_types = dict(src.dtypes)
  results = []
  for col, dtype in tgt.dtypes:
    if col in src_types and src_types[col] != dtype:
      results.append(CheckResult(
        f"dtype_{col}", "WARN",
        f"src={src_types[col]} tgt={dtype}"
      ))
  return results

# report_engine/report_builder.py
# Builds Pass/Fail validation reports

import json
from datetime import datetime
from typing import List
from ..validation_engine.row_count import CheckResult
from ..connectors.azure_conn import AzureADLSConnector

class ReportBuilder:
  """Builds structured validation reports with pass/fail
  status and detailed check results, stored to ADLS."""

  def __init__(self, results: List[CheckResult]):
    self.results = results
    self.adls = AzureADLSConnector()

  def build(self, run_id: str) -> dict:
    passed = [r for r in self.results if r.status == "PASS"]
    failed = [r for r in self.results if r.status == "FAIL"]
    warnings = [r for r in self.results if r.status == "WARN"]

    overall = "PASS" if not failed else (
      "WARN" if not failed else "FAIL"
    )
    return {
      "run_id": run_id,
      "timestamp": datetime.utcnow().isoformat(),
      "overall_status": overall,
      "summary": {
        "total": len(self.results),
        "passed": len(passed),
        "failed": len(failed),
        "warnings": len(warnings),
        "pass_rate": len(passed) / len(self.results)
      },
      "checks": [
        {"check": r.check, "status": r.status,
         "detail": r.detail}
        for r in self.results
      ]
    }

  def save(self, run_id: str):
    report = self.build(run_id)
    self.adls.write_report(json.dumps(report, indent=2), run_id)

from fastapi import FastAPI, BackgroundTasks, HTTPException

app = FastAPI(title="AI Data Validation API", version="1.0.0")

class RunRequest(BaseModel):
  source_type: str # oracle | hive | adls
  target_type: str
  tables: List[str]
  pk_columns: Optional[dict] = {}

@app.post("/api/v1/validate")
async def trigger_validation(req: RunRequest, bg: BackgroundTasks):
  """Trigger a new validation run — async background task."""
  run_id = f"RUN-{int(time.time())}"
  bg.add_task(run_validation_task, run_id, req)
  return {"run_id": run_id, "status": "queued"}

async def get_run_status(run_id: str):

  report = load_report_from_adls(run_id)
  if not report:
    raise HTTPException(404, "Run not found")
  return report

@app.get("/api/v1/runs")
async def list_runs(limit: int = 20):
  """List recent validation runs with summary stats."""
  return list_reports_from_adls(limit=limit)

async def health(): return {"status": "ok"}
